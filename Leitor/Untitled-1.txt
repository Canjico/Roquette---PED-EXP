#include <SFML/Graphics.hpp>

int main()
{
    // Cria uma janela de 800x600 pixels com título
    sf::RenderWindow window(sf::VideoMode(800, 600), "ComaVerso - Teste Inicial");

    // Loop principal (enquanto a janela estiver aberta)
    while (window.isOpen())
    {
        sf::Event event;
        while (window.pollEvent(event))
        {
            // Se clicar no X, fecha a janela
            if (event.type == sf::Event::Closed)
                window.close();
        }

        window.clear(sf::Color::Black); // limpa a tela (cor preta)
        window.display(); // mostra o que foi desenhado
    }

    return 0;
}
